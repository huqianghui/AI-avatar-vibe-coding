```tsx
export function AdminRoute() {
  const { user } = useAuthStore();
  if (user?.role !== "admin") return <Navigate to="/user/dashboard" replace />;  // UNCHANGED (D-10)
  return <Outlet />;
}

export function GuestRoute() {
  const { token, user } = useAuthStore();
  if (token) {
    const target =
      user?.role === "admin" ? "/admin/dashboard" : "/user/dashboard";  // <-- line 29, change else-branch to "/"
    return <Navigate to={target} replace />;
  }
  return <Outlet />;
}
```

    ```typescript
    import apiClient from "@/api/client";
    import type { CitationOut } from "@/types/avatar"; // reuse existing Citation type if present; else define locally: { title: string; url: string }

    export interface PersonalizedSessionResponse { session_id: string; expires_at: string }
    export interface PersonalizedChatResponse { answer: string; citations: CitationOut[]; is_refusal: boolean }

    export async function createPersonalizedSession(): Promise<PersonalizedSessionResponse> {
      const { data } = await apiClient.post<PersonalizedSessionResponse>("/avatar/session");

      sessionId: string,
      message: string,
      locale: string,
    ): Promise<PersonalizedChatResponse> {
      const { data } = await apiClient.post<PersonalizedChatResponse>("/avatar/chat", {
        session_id: sessionId,
        message,
        locale,
      });
      return data;
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser -> `/api/v1/avatar/session`, `/api/v1/avatar/chat` | Authenticated user's browser session, JWT auto-attached by `apiClient` interceptor |
| Frontend component tree -> DOM | Point where any leaked CRM data would become visible to the end user |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|------------------|
| T-33-12 | Information Disclosure | `avatar-page.tsx`, `personalized-avatar.ts` response types | mitigate | `PersonalizedChatResponse`/`PersonalizedSessionResponse` types carry only `answer`/`citations`/`is_refusal`/`session_id`/`expires_at` — no CRM field, preference tag, or match-status field exists anywhere in the frontend type surface or component props; Task 3's E2E test asserts seeded CRM strings never appear in rendered `innerText` |
| T-33-13 | Spoofing | `personalized-avatar.ts` API calls | mitigate | Uses only the shared `apiClient` singleton (JWT auto-attached from `localStorage` via existing interceptor) — no manual `user_id`/`session_id` override is ever accepted as a client-supplied identity claim; `session_id` sent in `PersonalizedChatRequest` is validated server-side against the JWT's `sub` via 33-05's `get_owned_session` IDOR gate, not trusted as an identity claim |
| T-33-14 | Tampering | Header badge / auth-state branch | accept | Badge visibility is purely `isAuthenticated`-driven UI cosmetics with zero security function — even if a user forged the visual badge client-side, all actual personalization data access remains gated server-side by JWT + `get_owned_session` (33-04/33-05); no client-side check is relied upon for authorization |
</threat_model>

    ```typescript
    import { useState, useCallback, useEffect } from "react";
    import { createAnonymousSession } from "@/api/public-avatar";

    export function useAnonymousAvatarSession() {
      const [sessionToken, setSessionToken] = useState<string | null>(null);
      const [expiresAt, setExpiresAt] = useState<string | null>(null);

      const renew = useCallback(async () => {
        const { sessionToken: token, expiresAt: exp } = await createAnonymousSession();
        setSessionToken(token);
        setExpiresAt(exp);
        return token;
      }, []);

      useEffect(() => { renew(); }, [renew]); // silent session creation on page load — fixed-window renewal (new session, no carried continuity), per 32-RESEARCH.md Open Question 3 resolution

      return { sessionToken, expiresAt, renew };
    }
    ```

    Create `frontend/src/hooks/use-anonymous-avatar-chat.ts`:
    ```typescript
    import { useMutation } from "@tanstack/react-query";
    import { sendAnonymousChat } from "@/api/public-avatar";

    export function useAnonymousAvatarChat(sessionToken: string | null, onSessionExpired: () => Promise<string>) {
      return useMutation({
        mutationFn: async (message: string) => {
          if (!sessionToken) throw new Error("no session");
          try {
            return await sendAnonymousChat(sessionToken, message);
          } catch (err) {
            if (err instanceof Response && err.status === 401) {
              const freshToken = await onSessionExpired();
              return await sendAnonymousChat(freshToken, message);
            }
            throw err;
          }
        },
      });
    }
    ```

```ts
type ButtonAppearance = "secondary" | "primary" | "outline" | "subtle" | "transparent";
declare const Button: React.ForwardRefExoticComponent<{
  appearance?: ButtonAppearance;
  size?: "small" | "medium" | "large";
  disabled?: boolean;
  icon?: React.ReactElement;
  iconPosition?: "before" | "after";
  shape?: "rounded" | "circular" | "square";
  className?: string;
  children?: React.ReactNode;
} & React.ButtonHTMLAttributes<HTMLButtonElement> & React.RefAttributes<HTMLButtonElement>>;
```

Mapping tables to implement exactly (per FEATURES.md, verified against Fluent's appearance/size enums):
```
variant -> appearance:      size -> size:
  default    -> primary       default -> medium
  destructive-> (Griffel override, see below)   sm      -> small
  outline    -> outline       lg      -> large
  secondary  -> secondary      icon    -> medium (square shape override; no dedicated Fluent icon-size token)
  ghost      -> subtle
  link       -> transparent (+ underline override via Griffel, matches existing `link` variant's text-underline styling)
```

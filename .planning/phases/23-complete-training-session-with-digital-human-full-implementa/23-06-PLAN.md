```python
"""Tests for session audio upload endpoint."""
import io
import pytest
from httpx import AsyncClient


@pytest.mark.asyncio
async def test_upload_session_audio_success(
    client: AsyncClient, auth_headers: dict, test_session_id: str
):
    """POST /sessions/{id}/audio uploads audio and returns URL."""
    audio_content = b"fake-webm-audio-content-for-testing"
    files = {"file": ("recording.webm", io.BytesIO(audio_content), "audio/webm")}
    
    response = await client.post(
        f"/api/v1/sessions/{test_session_id}/audio",
        files=files,
        headers=auth_headers,
    )
    assert response.status_code == 201
    data = response.json()
    assert "audio_url" in data
    assert data["voice_score_status"] == "pending"


@pytest.mark.asyncio
async def test_upload_audio_unauthorized(client: AsyncClient, test_session_id: str):
    """Audio upload requires authentication."""
    audio_content = b"fake-audio"
    files = {"file": ("recording.webm", io.BytesIO(audio_content), "audio/webm")}
    
    response = await client.post(
        f"/api/v1/sessions/{test_session_id}/audio",
        files=files,
    )
    assert response.status_code == 401


@pytest.mark.asyncio
async def test_upload_audio_wrong_user(
    client: AsyncClient, other_user_headers: dict, test_session_id: str
):
    """Cannot upload audio to another user's session."""
    audio_content = b"fake-audio"
    files = {"file": ("recording.webm", io.BytesIO(audio_content), "audio/webm")}
    
    response = await client.post(
        f"/api/v1/sessions/{test_session_id}/audio",
        files=files,
        headers=other_user_headers,
    )
    assert response.status_code == 403


@pytest.mark.asyncio
async def test_get_voice_score_status(
    client: AsyncClient, auth_headers: dict, test_session_id: str
):
    """GET /sessions/{id}/voice-score returns status."""
    response = await client.get(
        f"/api/v1/sessions/{test_session_id}/voice-score",
        headers=auth_headers,
    )
    assert response.status_code == 200
    data = response.json()
    assert data["session_id"] == test_session_id
    assert data["voice_score_status"] in ("none", "pending", "processing", "completed", "failed")
```

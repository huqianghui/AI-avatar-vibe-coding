Research Pattern 3 (hand-built Context, exact shape confirmed against 40-RESEARCH.md):
```tsx
const TabsContext = React.createContext<{ value: string; onValueChange: (v: string) => void } | null>(null);
function useTabsContext() {
  const ctx = React.useContext(TabsContext);
  if (!ctx) throw new Error("Tabs components must be used within <Tabs>");
  return ctx;
}
function Tabs({ value, onValueChange, defaultValue, children, ...props }: TabsProps) {
  // if uncontrolled (defaultValue only), manage internal state via useState, mirroring Radix's Root behavior
  return <TabsContext.Provider value={{ value, onValueChange }}>{children}</TabsContext.Provider>;
}
function TabsList({ children, ...props }: TabsListProps) {
  const ctx = useTabsContext();
  return (
    <FluentTabList selectedValue={ctx.value} onTabSelect={(_ev, data) => ctx.onValueChange(String(data.value))} {...props}>
      {children}
    </FluentTabList>
  );
}
function TabsTrigger({ value, children, ...props }: TabsTriggerProps) {
  const ctx = useTabsContext();
  const isActive = ctx.value === value;
  return (
    <FluentTab value={value} data-state={isActive ? "active" : "inactive"} {...props}>
      {children}
    </FluentTab>
  );
}
function TabsContent({ value, children, ...props }: TabsContentProps) {
  const ctx = useTabsContext();
  if (ctx.value !== value) return null;
  return (
    <div role="tabpanel" data-state="active" {...props}>
      {children}
    </div>
  );
}
```
Note: since `TabsContent` unmounts (returns null) when inactive, its own `data-state` is always `"active"` when rendered at all -- this matches the E2E specs' selector `[role='tabpanel'][data-state='active']` (they select the active panel, which is the only one in the DOM). Confirm this reasoning empirically via Test 3 below rather than assuming -- if any E2E spec expects an INACTIVE tabpanel to also exist in the DOM with `data-state="inactive"`, this conditional-unmount approach would break it; re-check `admin-dry-run.spec.ts:299` and `admin-skill-editor.spec.ts:103,337` context lines during read_first to confirm they only ever query the active panel.

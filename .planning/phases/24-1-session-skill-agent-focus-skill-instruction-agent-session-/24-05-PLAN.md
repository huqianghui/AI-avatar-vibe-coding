class TestExtractSopSteps:
    """Tests for SOP step extraction from various content formats."""

    def test_numbered_list(self):
        content = "1. Introduce yourself\n2. Present product data\n3. Handle objections\n4. Close"
        steps = extract_sop_steps(content)
        assert len(steps) == 4
        assert "Introduce yourself" in steps[0]
        assert "Close" in steps[3]

    def test_step_colon_format(self):
        content = "Step 1: Introduction\nStep 2: Data presentation\nStep 3: Q&A"
        steps = extract_sop_steps(content)
        assert len(steps) == 3
        assert "Introduction" in steps[0]

    def test_chinese_step_format(self):
        content = "步骤 1: 自我介绍\n步骤 2: 产品介绍\n步骤 3: 回答问题"
        steps = extract_sop_steps(content)
        assert len(steps) == 3
        assert "自我介绍" in steps[0]

    def test_fallback_paragraphs(self):
        content = "First topic here.\n\nSecond topic here.\n\nThird topic here."
        steps = extract_sop_steps(content)
        assert len(steps) == 3

    def test_empty_content(self):
        steps = extract_sop_steps("")
        assert steps == []

    def test_max_20_steps(self):
        content = "\n\n".join(f"Paragraph {i}" for i in range(30))

import pytest

from app.services.cu_evaluation_service import (
    build_content_analyzer_schema,
    build_voice_analyzer_schema,

class TestBuildContentAnalyzerSchema:
    """Tests for CU content analyzer fieldSchema generation."""

    def test_basic_dimensions(self):
        dims = [
            {"name": "Key Messages", "weight": 40, "criteria": ["Delivered all key points"], "max_score": 100},
            {"name": "Empathy", "weight": 30, "criteria": ["Showed understanding"], "max_score": 100},
        ]
        schema = build_content_analyzer_schema(dims)
        assert "fields" in schema
        # Should have dimension fields + feedback_summary
        assert "feedback_summary" in schema["fields"]
        # At least 2 dimension fields
        dim_fields = [k for k in schema["fields"] if k.startswith("dim_")]
        assert len(dim_fields) == 2

    def test_empty_dimensions(self):
        schema = build_content_analyzer_schema([])
        assert "fields" in schema
        assert "feedback_summary" in schema["fields"]

    def test_field_has_generate_method(self):
        dims = [{"name": "Test", "weight": 100, "criteria": [], "max_score": 100}]
        schema = build_content_analyzer_schema(dims)
        dim_field = [v for k, v in schema["fields"].items() if k.startswith("dim_")][0]
        assert dim_field["method"] == "generate"

class TestMergeScores:
    """Tests for layered score merging (D-11, D-12, D-13)."""

    def test_text_only_session(self):
        """D-13: Text-only = content only, 100% weight."""
        content = {
            "dimensions": [{"name": "km", "score": 85, "weight": 100}],
            "feedback_summary": "Good performance",
        }
        result = merge_scores(content, None, 60, 40)
        assert result["overall_score"] == 85.0
        assert result["voice_total"] is None
        assert result["content_total"] == 85.0

    def test_dual_mode_weighted_merge(self):
        """D-11: Layered merge with content 60% + voice 40%."""
        content = {
            "dimensions": [
                {"name": "km", "score": 80, "weight": 50},
                {"name": "empathy", "score": 90, "weight": 50},
            ],
            "feedback_summary": "Solid",
        }
        voice = {
            "dimensions": [
                {"name": "fluency", "score": 70, "weight": 50},
                {"name": "tone", "score": 60, "weight": 50},
            ],
            "feedback_summary": "Decent voice",
        }
        result = merge_scores(content, voice, 60, 40)
        # Content total: (80*50 + 90*50) / 100 = 85
        # Voice total: (70*50 + 60*50) / 100 = 65
        # Overall: 85*0.6 + 65*0.4 = 51 + 26 = 77
        assert result["content_total"] == 85.0
        assert result["voice_total"] == 65.0
        assert abs(result["overall_score"] - 77.0) < 0.1

    def test_custom_weights(self):
        """D-12: Admin can configure weights (e.g., 80/20)."""
        content = {"dimensions": [{"name": "a", "score": 100, "weight": 100}], "feedback_summary": ""}
        voice = {"dimensions": [{"name": "b", "score": 50, "weight": 100}], "feedback_summary": ""}
        result = merge_scores(content, voice, 80, 20)
        # 100*0.8 + 50*0.2 = 80 + 10 = 90
        assert abs(result["overall_score"] - 90.0) < 0.1

    def test_zero_voice_weight(self):
        """Edge case: voice_weight=0 means content only even with voice data."""
        content = {"dimensions": [{"name": "a", "score": 75, "weight": 100}], "feedback_summary": ""}
        voice = {"dimensions": [{"name": "b", "score": 50, "weight": 100}], "feedback_summary": ""}
        result = merge_scores(content, voice, 100, 0)
        assert result["overall_score"] == 75.0

    def test_includes_all_dimensions_in_output(self):
        """Result includes merged dimensions list."""
        content = {"dimensions": [{"name": "km", "score": 80, "weight": 100}], "feedback_summary": "ok"}
        voice = {"dimensions": [{"name": "fluency", "score": 70, "weight": 100}], "feedback_summary": "ok"}
        result = merge_scores(content, voice, 60, 40)
        assert len(result["dimensions"]) == 2

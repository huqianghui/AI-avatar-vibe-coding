This plan closes both gaps: it negotiates the video path AND wires persona identity through regardless of whether Azure's WebRTC-over-signaling-URL transport actually streams live avatar video for this session type (an open question already flagged for manual verification in 37-02's checkpoint task).
Output: `useAnonymousVoiceLive` returns `avatarCharacter`/`avatarStyle`/`isAvatarConnected`; `avatar-page.tsx` renders the correct persona identity; `persona-switch.spec.ts` proves the avatar identity (not just the switcher label) updates on switch.

```typescript
pc.ontrack = (event) => {
  if (event.track.kind === "video" && videoRef.current) {
    videoRef.current.srcObject = event.streams[0] ?? null;
    videoRef.current.play().catch((err: unknown) => {
      log.warn("Video play() failed: %o", err);
    });
  } else if (event.track.kind === "audio") {
    const audio = document.createElement("audio");
    audio.srcObject = event.streams[0] ?? null;
    audio.autoplay = true;
    audio.style.display = "none";
    document.body.appendChild(audio);
    audio.play().catch((err: unknown) => log.warn("Audio play() failed: %o", err));
  }
};
pc.addTransceiver("video", { direction: "recvonly" });
pc.addTransceiver("audio", { direction: "recvonly" });
```
